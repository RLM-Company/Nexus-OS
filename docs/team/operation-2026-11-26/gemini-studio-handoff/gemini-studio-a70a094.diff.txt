+  public readonly providerId = "external-image-provider";
+  public readonly mode = "image" as const;

+  private readonly baseUrl: string;
+  private readonly apiKey: string;
+  private readonly fetchImpl: typeof fetch;
+
+  constructor(options: {
+    baseUrl: string;
+    apiKey: string;
+    fetchImpl?: typeof fetch;
+  }) {
+    this.baseUrl = options.baseUrl.replace(/\/+$/, "");
+    this.apiKey = options.apiKey;
+    this.fetchImpl = options.fetchImpl ?? fetch;
+  }
+
+  public async startGeneration(req: ExternalMediaStartRequest): Promise<ExternalMediaProviderStatus> {
+    const res = await this.fetchImpl(`${this.baseUrl}/v1/images/generations`, {
+      method: "POST",
+      headers: {
+        "content-type": "application/json",
+        authorization: `Bearer ${this.apiKey}`,
+        "idempotency-key": req.idempotencyKey
+      },
+      body: JSON.stringify({
+        prompt: req.prompt,
+        aspect_ratio: req.parameters.aspectRatio ?? "1:1",
+        style: req.parameters.style ?? "editorial",
+        references: req.references
+      })
+    });
+
+    const data = (await res.json().catch(() => ({}))) as Record<string, any>;
+    if (!res.ok) {
+      return {
+        providerJobId: String(data.id ?? data.providerJobId ?? ""),
+        status: "failed",
+        errorCode: String(data.error?.code ?? data.errorCode ?? `IMAGE_PROVIDER_HTTP_${res.status}`),
+        errorMessage: String(
+          data.error?.message ?? data.errorMessage ?? `Ошибка провайдера изображений (HTTP ${res.status}).`
+        )
+      };
+    }
+
+    return {
+      providerJobId: String(data.id ?? data.providerJobId ?? ""),
+      status: (data.status as ExternalMediaProviderStatus["status"]) ?? "queued",
+      artifactUrl: data.artifact_url ?? data.artifactUrl ?? null,
+      mimeType: data.mime_type ?? data.mimeType ?? "image/png",
+      filename: data.filename ?? null,
+      errorCode: data.error_code ?? data.errorCode ?? null,
+      errorMessage: data.error_message ?? data.errorMessage ?? null
+    };
+  }
+
+  public async pollStatus(providerJobId: string): Promise<ExternalMediaProviderStatus> {
+    const res = await this.fetchImpl(
+      `${this.baseUrl}/v1/images/generations/${encodeURIComponent(providerJobId)}`,
+      {
+        method: "GET",
+        headers: { authorization: `Bearer ${this.apiKey}` }
+      }
+    );
+
+    const data = (await res.json().catch(() => ({}))) as Record<string, any>;
+    if (!res.ok) {
+      return {
+        providerJobId,
+        status: "failed",
+        errorCode: String(data.error?.code ?? data.errorCode ?? `IMAGE_PROVIDER_POLL_HTTP_${res.status}`),
+        errorMessage: String(
+          data.error?.message ?? data.errorMessage ?? `Ошибка проверки статуса изображения (HTTP ${res.status}).`
+        )
+      };
+    }
+
+    return {
+      providerJobId: String(data.id ?? data.providerJobId ?? providerJobId),
+      status: (data.status as ExternalMediaProviderStatus["status"]) ?? "running",
+      artifactUrl: data.artifact_url ?? data.artifactUrl ?? null,
+      mimeType: data.mime_type ?? data.mimeType ?? "image/png",
+      filename: data.filename ?? null,
+      errorCode: data.error_code ?? data.errorCode ?? null,
+      errorMessage: data.error_message ?? data.errorMessage ?? null
+    };
+  }
+
+  public async downloadArtifact(
+    artifactUrl: string,
+    fallbackFilename: string
+  ): Promise<ExternalMediaDownloadedArtifact> {
+    const fullUrl = artifactUrl.startsWith("http")
+      ? artifactUrl
+      : `${this.baseUrl}${artifactUrl.startsWith("/") ? "" : "/"}${artifactUrl}`;
+    const res = await this.fetchImpl(fullUrl, {
+      method: "GET",
+      headers: { authorization: `Bearer ${this.apiKey}` }
+    });
+    if (!res.ok) throw new Error(`IMAGE_PROVIDER_DOWNLOAD_HTTP_${res.status}`);
+    const buffer = Buffer.from(await res.arrayBuffer());
+    if (buffer.byteLength === 0) throw new Error("IMAGE_PROVIDER_ARTIFACT_EMPTY");
+    const mimeType = res.headers.get("content-type")?.split(";")[0]?.trim() || "image/png";
+    return { buffer, mimeType, filename: fallbackFilename };
+  }

+  const server = http.createServer(async (req, res) => {
+    const parsedUrl = new URL(req.url ?? "/", "http://127.0.0.1");
+    const method = (req.method ?? "GET").toUpperCase();
+    const match = routes.find((r) => r.method === method && r.regex.test(parsedUrl.pathname));
+    if (!match) {
+      res.statusCode = 404;
+      res.end(JSON.stringify({ error: { code: "NOT_FOUND" } }));
+      return;
+    }
+    const params = (parsedUrl.pathname.match(match.regex)?.groups ?? {}) as Record<string, string>;
+    const query: Record<string, string> = {};
+    for (const [k, v] of parsedUrl.searchParams.entries()) query[k] = v;
+
+    let body: unknown = undefined;
+    if (method === "POST") {
+      const chunks: Buffer[] = [];
+      for await (const c of req) chunks.push(Buffer.isBuffer(c) ? c : Buffer.from(c));
+      const raw = Buffer.concat(chunks).toString("utf8");
+      if (raw) body = JSON.parse(raw);
+    }
+
+    const headers: Record<string, string> = {};
+    let statusCode = 200;
+    const reply = {
+      status(code: number) {
+        statusCode = code;
+        return this;
+      },
+      header(k: string, v: string) {
+        headers[k.toLowerCase()] = v;
+        return this;
+      },
+      send(payload: unknown) {
+        res.statusCode = statusCode;
+        for (const [k, v] of Object.entries(headers)) res.setHeader(k, v);
+        if (Buffer.isBuffer(payload)) res.end(payload);
+        else {
+          if (!headers["content-type"]) res.setHeader("content-type", "application/json; charset=utf-8");
+          res.end(JSON.stringify(payload));
+        }
+        return this;
+      }
+    };
+    await match.handler({ headers: req.headers, params, query, body }, reply);
+  });
+
+  return {
+    async start(): Promise<string> {
+      await new Promise<void>((r) => server.listen(0, "127.0.0.1", () => r()));
+      return `http://127.0.0.1:${(server.address() as { port: number }).port}`;
+    },
+    async close(): Promise<void> {
+      await new Promise<void>((r) => { server.closeAllConnections?.(); server.close(() => r()); });
+    }
+  };

+const temp = mkdtempSync(join(tmpdir(), "kosmora-5cycles-"));
+
+function emit(sourceRel, targetRel) {
+  let code = ts.transpileModule(readFileSync(resolve(root, sourceRel), "utf8"), {
+    fileName: sourceRel,
+    compilerOptions: { module: ts.ModuleKind.ESNext, target: ts.ScriptTarget.ES2022 }
+  }).outputText;
+  const sqliteShimUrl = pathToFileURL(resolve(root, "scripts/offline-typescript.mjs")).href;
+  code = code.replace(/from\s+["']node:sqlite["']/g, `from "${sqliteShimUrl}"`);
+  code = code.replace(/(from\s+["']\.[^"']+)\.ts(["'])/g, "$1.js$2");
+  code = code.replace(
+    /require\(["']\.\.\/\.\.\/\.\.\/\.\.\/desktop\/windows\//g,
+    `require(${JSON.stringify(resolve(root, "desktop/windows") + "/")}.replace(/\\/$/, "") + "/`
+  );
+  const outPath = join(temp, targetRel);
+  mkdirSync(dirname(outPath), { recursive: true });
+  writeFileSync(outPath, code, "utf8");
+}
+
+function emitDir(relDir) {
+  for (const name of readdirSync(resolve(root, relDir))) {
+    if (!name.endsWith(".ts")) continue;
+    const rel = `${relDir}/${name}`;
+    emit(rel, rel.replace(/\.ts$/, ".js"));
+  }
+}

+  );
